"""Readout-profiled loss and coupled versus diagonal coefficient corrections.

No teacher features enter these calculations. Features include the network's
1/sqrt(width) normalization; columns may include a learned constant intercept.
"""

import torch


def profile_readout(features, targets):
    """Minimum-norm least-squares readout and its empirical half-MSE.

    SVD handles rank deficiency. The envelope gradient is valid on a locally
    constant-rank feature map; rank changes need finite-change evaluation.
    """
    coefficients = torch.linalg.lstsq(features, targets, driver="gelsd").solution
    residual = targets - features @ coefficients
    return coefficients, .5 * residual.square().mean()


def diagonal_correction(features, targets, coefficients):
    """One diagonally preconditioned readout step with exact line search.

    Uses the same full training data as profiling, but does not invert the
    off-diagonal Gram coupling. The scalar line search includes that coupling
    in the denominator to guarantee non-increasing fixed-feature squared loss.
    """
    residual = targets - features @ coefficients
    work = features.T @ residual
    direction = work / features.square().sum(0).clamp_min(1e-30)
    movement = features @ direction
    scale = (residual @ movement) / movement.square().sum().clamp_min(1e-30)
    return coefficients + scale * direction
