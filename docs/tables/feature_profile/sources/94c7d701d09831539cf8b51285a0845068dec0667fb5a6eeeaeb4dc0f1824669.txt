#!/usr/bin/env python3
"""Paired interventions on coupled readout learning during feature formation."""

import argparse
import hashlib
import json
import math
import time
from pathlib import Path

import pandas as pd
import torch

from neural_competition.feature_profile import diagonal_correction, profile_readout
from neural_competition.tasks import paired_gaussian_task
from neural_competition.training import advance, batch_gradient, components

ROOT = Path(__file__).resolve().parents[2]
ARMS = ["ordinary8", "ordinary32", "profile", "diagonal", "early", "late", "fast_readout"]


def features(p, x, width, bias):
    phi = (x @ p[:width, :20].T + p[:width, -2]).relu() / math.sqrt(width)
    if bias:
        phi = torch.cat([phi, torch.ones(len(x), 1, dtype=x.dtype) / math.sqrt(width)], 1)
    return phi


def coefficients(p, width, bias):
    return torch.cat([p[:width, -1], p[32:33, -1]]) if bias else p[:width, -1]


def set_coefficients(p, a, width, bias):
    p[:width, -1] = a[:width]
    if bias:
        p[32, -1] = a[-1]


def run(args):
    start = time.monotonic()
    folder = ROOT / "runs/feature_profile" / args.collection
    folder.mkdir(parents=True, exist_ok=True)
    tag = f"d{args.data_seed}_a{args.degrees}_r{args.ratio:g}_{args.optimizer}_{args.bias}"
    marker = folder / f"{tag}.json"
    if marker.exists():
        assert json.loads(marker.read_text())["args"] == vars(args), "Use a new collection"
        print("SKIP", tag, flush=True)
        return
    sources = [
        Path(__file__),
        ROOT / "src/neural_competition/feature_profile.py",
        ROOT / "src/neural_competition/training.py",
        ROOT / "src/neural_competition/tasks.py",
        ROOT / "src/neural_competition/residual_growth.py",
        ROOT / "docs/theory/feature_profile_protocol.md",
    ]
    hashes = {str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in sources}
    snapshots = folder / "sources"
    snapshots.mkdir(exist_ok=True)
    for p in sources:
        (snapshots / (hashes[str(p.relative_to(ROOT))] + ".txt")).write_bytes(p.read_bytes())
    d = paired_gaussian_task(args.degrees, args.ratio, args.data_seed)
    x, y, xt, yt = (d[k] for k in ["xtrain", "ytrain", "xtest", "ytest"])
    bias = args.bias != "none"
    models = [
        dict(seed=s, width=32 if arm == "ordinary32" else 8, arm=arm)
        for s in range(args.seed_start, args.seed_start + args.seeds)
        for arm in ARMS
    ]
    n = len(models)
    p = torch.zeros(n, 33, 22, dtype=torch.float64)
    norm = torch.tensor([[math.sqrt(m["width"])] for m in models], dtype=torch.float64)
    rates = torch.ones(n, 33, 22, dtype=torch.float64)
    kind = ["sgd", "momentum", "adam"].index(args.optimizer)
    for i, m in enumerate(models):
        s, w = m["seed"], m["width"]
        p[i, :w, :20] = torch.randn(
            w, 20, generator=torch.Generator().manual_seed(93000 + s), dtype=torch.float64
        ) / math.sqrt(20)
        p[i, :w, -1] = 0.1 * torch.randn(
            w, generator=torch.Generator().manual_seed(94000 + s), dtype=torch.float64
        )
        rates[i, w:] = 0
        if bias:
            p[i, 32, -2] = 1
            rates[i, 32, -1] = math.sqrt(w) if kind == 2 else w
            p[i, 32, -1] = norm[i, 0] * y.mean()
        if m["arm"] == "fast_readout":
            rates[i, :, -1] *= 10
    first, second = torch.zeros_like(p), torch.zeros_like(p)
    seed_map = torch.tensor([m["seed"] - args.seed_start for m in models])
    rngs = [
        torch.Generator().manual_seed(95000 + s)
        for s in range(args.seed_start, args.seed_start + args.seeds)
    ]
    lr = [0.3, 0.03, 0.003][kind]
    checkpoints = sorted(
        set(
            t
            for t in [0, 10, 100, 1000, 3000, args.steps - 1000, args.steps]
            if 0 <= t <= args.steps
        )
    )
    rows, corrections = [], []
    saved_states = {}
    for step in range(0, args.steps + 1, 10):
        if step in checkpoints:
            # States before scheduled corrections; all arms get the same terminal diagnostic.
            preds = {
                v: components(p, d["x" + v][None].expand(n, -1, -1), norm).sum(-1)
                for v in ["train", "test"]
            }
            for i, m in enumerate(models):
                w = m["width"]
                phi = features(p[i], x, w, bias)
                a = coefficients(p[i], w, bias)
                astar, profiled = profile_readout(phi, y)
                phit = features(p[i], xt, w, bias)
                sv = torch.linalg.svdvals(phi)
                q = p[i : i + 1].clone()
                set_coefficients(q[0], astar, w, bias)
                # Same full-objective Euclidean gradients; not Adam/momentum actual directions.
                gp = batch_gradient(q, x[None], y[None], norm[i : i + 1])[0, :w, :20]
                ga = batch_gradient(p[i : i + 1], x[None], y[None], norm[i : i + 1])[0, :w, :20]
                loss = float(0.5 * (preds["train"][i] - y).square().mean())
                rows.append(
                    dict(
                        **m,
                        step=step,
                        train_loss=loss,
                        test_loss=float(0.5 * (preds["test"][i] - yt).square().mean()),
                        profiled_train_loss=float(profiled),
                        profiled_test_loss=float(0.5 * (phit @ astar - yt).square().mean()),
                        equilibration_gap=loss - float(profiled),
                        gram_condition=float((sv[0] / sv[-1].clamp_min(1e-30)) ** 2),
                        positive=int((a[:w] > 0).sum()),
                        optimal_positive=int((astar[:w] > 0).sum()),
                        sign_disagreement=int(((astar[:w] * a[:w]) < 0).sum()),
                        profile_gradient_work=float((gp * ga).sum()),
                        profile_gradient_norm2=float(gp.square().sum()),
                    )
                )
            if step in [0, 1000, args.steps - 1000, args.steps]:
                saved_states[step] = dict(
                    parameters=p.clone(), first=first.clone(), second=second.clone()
                )
        if step == args.steps:
            break
        for i, m in enumerate(models):
            arm = m["arm"]
            active = (
                arm in ["profile", "diagonal"]
                or (arm == "early" and step < 1000)
                or (arm == "late" and step >= args.steps - 1000)
            )
            if not active:
                continue
            w = m["width"]
            phi = features(p[i], x, w, bias)
            a = coefficients(p[i], w, bias).clone()
            updated = (
                diagonal_correction(phi, y, a) if arm == "diagonal" else profile_readout(phi, y)[0]
            )
            set_coefficients(p[i], updated, w, bias)
            first[i, :, -1] = 0
            second[i, :, -1] = 0
            if step in checkpoints:
                corrections.append(
                    dict(
                        **m,
                        step=step,
                        before=float(0.5 * (phi @ a - y).square().mean()),
                        after=float(0.5 * (phi @ updated - y).square().mean()),
                        displacement=float((phi @ (updated - a)).square().mean().sqrt()),
                    )
                )
        count = min(10, args.steps - step)
        ids = torch.stack([torch.randint(4096, (count, 128), generator=g) for g in rngs], 1)
        p, first, second = advance(
            p, first, second, x, y, ids, seed_map, norm, rates, lr, kind, False, step
        )
    pd.DataFrame(rows).to_csv(
        folder / f"{tag}.csv.gz",
        index=False,
        float_format="%.17g",
        compression={"method": "gzip", "mtime": 0},
    )
    pd.DataFrame(corrections).to_csv(folder / f"{tag}_corrections.csv", index=False)
    torch.save(dict(states=saved_states, models=models), folder / f"{tag}.pt")
    manifest = dict(
        args=vars(args),
        sources=hashes,
        elapsed_seconds=time.monotonic() - start,
        dataset_hash=hashlib.sha256(x.numpy().tobytes() + y.numpy().tobytes()).hexdigest(),
    )
    marker.write_text(json.dumps(manifest, indent=2) + "\n")
    final = pd.DataFrame(rows).query("step==@args.steps").groupby("arm").test_loss.mean()
    print(tag, round(time.monotonic() - start, 1), final.to_dict(), flush=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--degrees", type=int, default=30)
    parser.add_argument("--ratio", type=float, default=-3)
    parser.add_argument("--data-seed", type=int, default=9027)
    parser.add_argument("--bias", choices=["none", "centered"], default="none")
    parser.add_argument("--optimizer", choices=["sgd", "momentum", "adam"], default="sgd")
    parser.add_argument("--seed-start", type=int, default=100)
    parser.add_argument("--seeds", type=int, default=10)
    parser.add_argument("--steps", type=int, default=10000)
    parser.add_argument("--collection", default="main")
    args = parser.parse_args()
    assert args.steps >= 1000 and args.steps % 10 == 0
    torch.set_num_threads(1)
    torch.set_num_interop_threads(1)
    run(args)


if __name__ == "__main__":
    main()
