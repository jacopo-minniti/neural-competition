#!/usr/bin/env python3
"""Audit and summarize all sign-survival cohorts, including population certificates."""

import hashlib
import json
import math
import shutil
from pathlib import Path

import matplotlib

matplotlib.use('Agg')
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
import torch

from neural_competition.hermite_certificate import population_certificate
from neural_competition.tasks import paired_gaussian_task
from neural_competition.training import components

ROOT = Path(__file__).resolve().parents[2]


def interval(values):
    rng = np.random.default_rng(421)
    return np.quantile(rng.choice(values, (5000, len(values))).mean(1), [.025, .975])


def main():
    torch.set_num_threads(1)
    output = ROOT/'docs/tables/sign_survival'
    output.mkdir(parents=True, exist_ok=True)
    snapshots = output/'sources'
    snapshots.mkdir(exist_ok=True)
    curves, certified, provenance, audits = [], [], [], []
    for marker in sorted((ROOT/'runs/sign_survival').glob('*/*.json')):
        manifest = json.loads(marker.read_text())
        args = manifest['args']
        collection = marker.parent.name
        provenance.append(dict(collection=collection, tag=marker.stem, **manifest))
        for source, digest in manifest['sources'].items():
            path = marker.parent/'sources'/f'{digest}.txt'
            assert hashlib.sha256(path.read_bytes()).hexdigest()==digest, source
            shutil.copyfile(path, snapshots/path.name)
        df = pd.read_csv(marker.with_suffix('.csv.gz'))
        df['collection'] = collection
        curves.append(df)
        bundle = torch.load(marker.with_suffix('.pt'), weights_only=False)
        d = paired_gaussian_task(args['degrees'], args['ratio'], args['data_seed'])
        assert manifest['dataset_hash']==hashlib.sha256(
            d['xtrain'].numpy().tobytes()+d['ytrain'].numpy().tobytes()).hexdigest()
        last = args['steps']
        p = bundle['states'][last]['parameters']
        models = bundle['models']
        norm = torch.tensor([[math.sqrt(m['width'])] for m in models], dtype=torch.float64)
        pred = components(p, d['xtest'][None].expand(len(models), -1, -1), norm).sum(-1)
        actual = .5*(pred-d['ytest']).square().mean(1)
        saved = df.query('step==@last').test_loss.to_numpy()
        np.testing.assert_allclose(actual.numpy(), saved, rtol=1e-10, atol=1e-13)
        # Late control and baseline must agree exactly before intervention.
        checkpoint = last-1000
        for i, m in enumerate(models):
            if m['arm']=='protect_late':
                j = next(j for j, other in enumerate(models)
                         if other['seed']==m['seed'] and other['arm']=='balanced8')
                for key in ['parameters', 'first', 'second']:
                    assert torch.equal(bundle['states'][checkpoint][key][i],
                                       bundle['states'][checkpoint][key][j])
            w = m['width']
            cert = population_certificate(d['teacher'], d['beta'], p[i, :w, :20],
                                          p[i, :w, -1]/math.sqrt(w))
            assert cert['population_mse']+1e-10 >= cert['quadratic_error']
            assert cert['quadratic_error']+1e-10 >= cert['sign_bound']
            assert cert['higher_error'] >= -1e-10
            certified.append(dict(collection=collection, optimizer=manifest['optimizer'],
                                  **m, **{k:v for k,v in cert.items() if k!='positive'}))
        if collection=='movement_control':
            reference = torch.load(ROOT/'runs/sign_survival/main'/marker.with_suffix('.pt').name,
                                   weights_only=False)
            for i,m in enumerate(models):
                if m['arm']=='damp_early':
                    continue
                j = next(j for j, other in enumerate(reference['models']) if m==other)
                for step in reference['states']:
                    for key in ['parameters', 'first', 'second']:
                        assert torch.equal(bundle['states'][step][key][i],
                                           reference['states'][step][key][j])
        audits.append(dict(collection=collection, optimizer=manifest['optimizer'],
                           models=len(models), max_loss_error=float(np.max(abs(actual.numpy()-saved))),
                           failures=len(manifest['failures'])))
    frame = pd.concat(curves, ignore_index=True)
    frame.to_csv(output/'trajectories.csv.gz', index=False,
                 compression={'method':'gzip', 'mtime':0})
    certificates = pd.DataFrame(certified)
    certificates.to_csv(output/'population_certificates.csv', index=False)
    final = frame.loc[frame.step==frame.groupby(['collection','optimizer']).step.transform('max')]
    final.to_csv(output/'endpoints.csv', index=False)
    summary = final.groupby(['collection','optimizer','arm']).agg(
        seeds=('seed','count'), mean_loss=('test_loss','mean'), median_loss=('test_loss','median'),
        profiled_loss=('profiled_test_loss','mean'), mean_positive=('positive','mean'),
        mean_zero=('zero','mean'), projection_count=('projection_count','mean'),
        correction_l1=('correction_l1','mean'), correction_l2_sum=('correction_l2_sum','mean'),
        damping_caps=('damping_caps','mean')).reset_index()
    summary.to_csv(output/'summary.csv', index=False)
    contrasts = []
    for (collection, optimizer), group in final.groupby(['collection','optimizer']):
        table = group.pivot(index='seed', columns='arm', values='test_loss')
        for arm in table.columns:
            if arm=='balanced8':
                continue
            delta = 200*(table.balanced8-table[arm]).to_numpy()
            lo,hi = interval(delta)
            contrasts.append(dict(collection=collection, optimizer=optimizer, arm=arm,
                                  seeds=len(delta), improvement_pp=delta.mean(), low_pp=lo, high_pp=hi,
                                  wins=int((delta>0).sum()),
                                  gap_closed=(table.balanced8-table[arm]).mean()
                                  /(table.balanced8-table.ordinary32).mean()))
        if 'damp_early' in table:
            delta=200*(table.damp_early-table.protect_early).to_numpy()
            lo,hi=interval(delta)
            contrasts.append(dict(collection=collection, optimizer=optimizer,
                                  arm='protect_early_vs_damping', seeds=len(delta),
                                  improvement_pp=delta.mean(), low_pp=lo, high_pp=hi,
                                  wins=int((delta>0).sum())))
    pd.DataFrame(contrasts).to_csv(output/'contrasts.csv', index=False)
    pd.DataFrame(audits).to_csv(output/'audit.csv', index=False)
    (output/'provenance.json').write_text(json.dumps(provenance, indent=2)+'\n')
    analysis_sources=[Path(__file__), ROOT/'src/neural_competition/hermite_certificate.py']
    analysis_hashes={str(p.relative_to(ROOT)):hashlib.sha256(p.read_bytes()).hexdigest()
                     for p in analysis_sources}
    for path in analysis_sources:
        (snapshots/(analysis_hashes[str(path.relative_to(ROOT))]+'.txt')).write_bytes(path.read_bytes())
    (output/'analysis_provenance.json').write_text(json.dumps(analysis_hashes,indent=2)+'\n')
    figs=ROOT/'docs/figures/sign_survival'
    figs.mkdir(parents=True, exist_ok=True)
    fig,axes=plt.subplots(1,3,figsize=(12,3.7),constrained_layout=True)
    arms=['balanced8','protect_early','protect_all','profile_early','ordinary32']
    labels=['Ordinary\n8, balanced','Early sign\nprotection','Always sign\nprotection',
            'Early\nreadout fit','Ordinary\n32']
    colors=['#737373','#167d9a','#38a87c','#ae7951','#8064a2']
    for ax,opt in zip(axes,['sgd','momentum','adam']):
        g=final.query('collection=="main" and optimizer==@opt')
        for j,(arm,color) in enumerate(zip(arms,colors)):
            a=200*g.loc[g.arm==arm,'test_loss']
            ax.scatter(np.linspace(j-.12,j+.12,len(a)),a.clip(lower=1e-7),s=12,
                       color=color,alpha=.55)
            ax.plot([j-.22,j+.22],[a.mean(),a.mean()],color=color,lw=3)
            ax.text(j, 9, f'{a.mean():.3g}%',ha='center',fontsize=8)
        ax.set(yscale='log',ylim=(1e-7,16),xticks=range(5),xticklabels=labels,
               title=opt.upper() if opt=='sgd' else opt.capitalize())
        ax.tick_params(axis='x',labelsize=8)
        ax.grid(axis='y',alpha=.15)
    axes[0].set_ylabel('Test MSE / training target variance (%)')
    fig.suptitle('Blocking sign crossings early rescues narrow feature learning\n20 paired starts; bars: means; dots: starts (display floor 10⁻⁷%)',fontsize=12)
    fig.savefig(figs/'interventions.png',dpi=180)
    plt.close(fig)
    fig,axes=plt.subplots(1,3,figsize=(11,3.5),constrained_layout=True)
    for ax,opt in zip(axes,['sgd','momentum','adam']):
        g=certificates.query('collection=="main" and optimizer==@opt and arm=="balanced8"')
        ax.scatter(100*g.sign_bound,100*g.population_mse,color='#167d9a')
        high=max(100*g.population_mse.max(),3)
        ax.plot([0,high],[0,high],ls='--',color='#888888',lw=1)
        ax.set(xlabel='Certified minimum population MSE (%)',title=opt.capitalize(),
               xlim=(-.1,high),ylim=(-.1,high))
        ax.grid(alpha=.15)
    axes[0].set_ylabel('Actual population MSE (%)')
    fig.suptitle('Missing signed roles impose an approximation error floor',fontsize=12)
    fig.savefig(figs/'certificate.png',dpi=180)
    plt.close(fig)
    print(summary.query('collection=="main"').to_string(index=False))
    print(pd.DataFrame(contrasts).query('collection=="main"').to_string(index=False))
    print('Audited',len(audits),'bundles;',len(final),'endpoints')


if __name__=='__main__':
    main()
